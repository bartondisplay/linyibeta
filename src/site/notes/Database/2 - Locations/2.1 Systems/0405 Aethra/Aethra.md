# 0405 Aethra

![Aethra.jpg](/img/user/Images/Aethra.jpg)
*The Aethra system from nearby Scalar*

> [!summary]
> Aethra is a star system in the [[Database/2 - Locations/Central Belt\|central belt]] of [[Database/Linyi Beta\|Linyi Beta]]. It is one of the most developed systems in the sector. The planets [[Database/2 - Locations/2.1 Systems/0405 Aethra/Bayron\|Bayron]], [[Database/2 - Locations/2.1 Systems/0405 Aethra/Scalar\|Scalar]], [[Database/2 - Locations/2.1 Systems/0405 Aethra/Kostan\|Kostan]] and [[Database/2 - Locations/2.1 Systems/0405 Aethra/Katas\|Katas]] are the system’s planetary bodies. Other locations include industrial zones with habitation centres, such as [[Database/2 - Locations/2.1 Systems/0405 Aethra/006 Stantar's Rift\|Stantar's Rift]], [[Database/2 - Locations/2.1 Systems/0405 Aethra/008 Untriar Debris Field\|Untriar Debris Field]] and [[Database/2 - Locations/2.1 Systems/0405 Aethra/326 Thiï'dri Nebula\|Thiï'dri Nebula]]. Travelers arriving from neighbouring systems will report to [[Database/2 - Locations/2.1 Systems/0405 Aethra/Warpheart Astrastation\|Warpheart Astrastation]], the system’s instra-space station.

(function() {
  var svg = document.getElementById('aethra-svg');
  if (!svg) return;
  var baseVB = [-1266.3302856445314, -1266.3302856445314, 2532.660571289063, 2532.660571289063];
  var vb = baseVB.slice();
  var isPanning = false;
  var start = {x:0, y:0};

  function apply() {
    svg.setAttribute('viewBox', vb.join(' '));
  }

  function zoom(factor) {
    var cx = vb[0] + vb[2] / 2;
    var cy = vb[1] + vb[3] / 2;
    var newW = vb[2] * factor;
    var newH = vb[3] * factor;
    newW = Math.max(200, Math.min(newW, baseVB[2] * 4));
    newH = Math.max(200, Math.min(newH, baseVB[3] * 4));
    vb[0] = cx - newW / 2;
    vb[1] = cy - newH / 2;
    vb[2] = newW;
    vb[3] = newH;
    apply();
  }

  document.getElementById('aethra-zoom-in').addEventListener('click', function(){ zoom(0.8); });
  document.getElementById('aethra-zoom-out').addEventListener('click', function(){ zoom(1.25); });
  document.getElementById('aethra-zoom-reset').addEventListener('click', function(){ vb = baseVB.slice(); apply(); });

  svg.addEventListener('pointerdown', function(e) {
    isPanning = true;
    start.x = e.clientX;
    start.y = e.clientY;
    svg.style.cursor = 'grabbing';
    svg.setPointerCapture(e.pointerId);
  });
  svg.addEventListener('pointermove', function(e) {
    if (!isPanning) return;
    var scale = vb[2] / svg.clientWidth;
    var dx = (e.clientX - start.x) * scale;
    var dy = (e.clientY - start.y) * scale;
    vb[0] -= dx;
    vb[1] -= dy;
    start.x = e.clientX;
    start.y = e.clientY;
    apply();
  });
  function endPan(e) {
    isPanning = false;
    svg.style.cursor = 'grab';
  }
  svg.addEventListener('pointerup', endPan);
  svg.addEventListener('pointerleave', endPan);

  svg.addEventListener('wheel', function(e) {
    e.preventDefault();
    zoom(e.deltaY > 0 ? 1.1 : 0.9);
  }, {passive:false});
})();
